57static void TECsTask(void *pvParameter)
58{
59 while(true)
60 {
61 uint8_t teclas = SwitchesRead();
62
63 // Control de TEC1 (Cambiar Estado MEDICION)
64 if(teclas & SWITCH_1)
65 {
66 MEDICION = !MEDICION;
67 }
68 // Control de TEC2 (Cambiar Estado HOLD)
69 if(teclas & SWITCH_2)
70 {
71 HOLD = !HOLD;
72 }
73 vTaskDelay(CONFIG_BLINK_PERIOD_TECS_MS / portTICK_PERIOD_MS);
74 }

91static void LedsyLcdTask(void *pvParameter)
92{
93 while(true)
94 {
95 if(MEDICION)
96 {
97 // Control de LEDs por rangos de distancia
98 if(DISTANCIA < 10)
99 {
100 LedsOffAll();
101 }
102 else if(DISTANCIA >= 10 && DISTANCIA < 20)
103 {
104 LedOn(LED_1);
105 LedOff(LED_2);
106 LedOff(LED_3);
107 }
108 else if(DISTANCIA >= 20 && DISTANCIA < 30)
109 {
110 LedOn(LED_1);
111 LedOn(LED_2);
112 LedOff(LED_3);
113 }
114 else if(DISTANCIA >= 30)
115 {
116 LedOn(LED_1);
117 LedOn(LED_2);
118 LedOn(LED_3);
119 }
120
121 // Control del Display LCD
122 if(!HOLD)
123 {
124 LcdItsE0803Write(DISTANCIA);
125 }
126 }
127 else
128 {
129 // Si MEDICION == No, apagar LEDs y limpiar LCD
130 LedsOffAll();
131 LcdItsE0803Off();
132 }
133
134 // Refresco de medición: 1 segundo
135 vTaskDelay(CONFIG_BLINK_PERIOD_REFRESH_MS / portTICK_PERIOD_MS);
136 }
137}
