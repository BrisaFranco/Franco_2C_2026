151static int16_t MedirReposo(void) {
152 double suma = 0.0, suma_cuad = 0.0;
153
154 IniciarMedicion();
155
156 for (uint32_t i = 0; i < MUESTRAS_CALIB; i++) {
157 EsperarTick();
158 uint16_t raw = 0;
159 float filtrada = LeerMuestraFiltrada(&raw);
160
161 suma += (double)filtrada;
162 suma_cuad += (double)filtrada * (double)filtrada;
163
164 if (((i + 1) % SAMPLE_FREQ) == 0) {
165 printf(" Segundos: %lu/%d (ADC crudo: %u | filtrada: %.1f)\r\n",
166 (unsigned long)((i + 1) / SAMPLE_FREQ), CALIB_SEC, (unsigned)raw, filtrada);
167 }
168 }
169
170 DetenerMuestreo();
171
172 double media = suma / (double)MUESTRAS_CALIB;
173 double varianza = suma_cuad / (double)MUESTRAS_CALIB - media * media;
174 if (varianza < 0.0) varianza = 0.0;
175 ruido_sigma = (float)sqrt(varianza);
176
177 printf("--> Ruido en reposo (sigma): %.1f cuentas\r\n", ruido_sigma);
178 return (int16_t)media;
179}

187static int16_t MedirPicos(int signo, const char *etiqueta) {
188 float umbral_alto = K_RUIDO_DETECCION * ruido_sigma;
189 if (umbral_alto < UMBRAL_DET_MIN_CUENTAS) umbral_alto = UMBRAL_DET_MIN_CUENTAS;
190 const float umbral_bajo = umbral_alto * 0.5f;
191
192 bool en_pico = false;
193 float pico_actual = 0.0f;
194 float extremo_global = 0.0f;
195 float pico_menor = 0.0f, pico_mayor = 0.0f;
196 double suma_picos = 0.0;
197 uint32_t n_picos = 0;
198
199 printf("--> Umbral de detección de picos: %.1f cuentas\r\n", umbral_alto);
200
201 IniciarMedicion();
202
203 for (uint32_t i = 0; i < MUESTRAS_CALIB; i++) {
204 EsperarTick();
205 uint16_t raw = 0;
206 float filtrada = LeerMuestraFiltrada(&raw);
207 float s = (float)signo * (filtrada - (float)offset_v);
208
209 if (s > extremo_global) extremo_global = s;
210
211 if (!en_pico) {
212 if (s > umbral_alto) {
213 en_pico = true;
214 pico_actual = s;
215 }
216 } else {
217 if (s > pico_actual) pico_actual = s;
218 if (s < umbral_bajo) {
219 if (n_picos == 0 || pico_actual < pico_menor) pico_menor = pico_actual;
220 if (n_picos == 0 || pico_actual > pico_mayor) pico_mayor = pico_actual;
221 suma_picos += (double)pico_actual;
222 n_picos++;
223 en_pico = false;
224 }
225 }
226
227 if (((i + 1) % SAMPLE_FREQ) == 0) {
228 printf(" Segundos: %lu/%d (ADC crudo: %u | picos: %lu | promedio: %.1f)\r\n",
229 (unsigned long)((i + 1) / SAMPLE_FREQ), CALIB_SEC, (unsigned)raw,
230 (unsigned long)n_picos, (n_picos > 0) ? (suma_picos / n_picos) : 0.0);
231 }
232 }
233
234 DetenerMuestreo();
235
236 float promedio;
237 if (n_picos >= PICOS_MIN_VALIDOS) {
238 promedio = (float)(suma_picos / (double)n_picos);
239 printf("--> %s: %lu picos | menor: %.1f | mayor: %.1f | PROMEDIO: %.1f cuentas\r\n",
240 etiqueta, (unsigned long)n_picos, pico_menor, pico_mayor, promedio);
241 } else {
242 promedio = extremo_global;
243 printf("!! ADVERTENCIA (%s): solo %lu picos detectados (mínimo %d). "
244 "Se usa el máximo absoluto (%.1f). Repita con movimientos más marcados.\r\n",
245 etiqueta, (unsigned long)n_picos, PICOS_MIN_VALIDOS, promedio);
246 }
247
248 return (int16_t)((float)signo * promedio);
249}

334static void UpdateMouse(int8_t *pos, uint8_t *click_flag, int16_t eog_centrado) {
335 static uint16_t t_estado = 0;
336 static uint16_t t_banda = 0;
337
338 *click_flag = 0;
339 *pos = 0;
340
341 if (!calibracion_completada) return;
342
343 const bool arriba_ok = (umbral_arriba_ref > 0);
344 const bool abajo_ok = (umbral_abajo_ref < 0);
345 const bool pestaneo_ok = (umbral_pestaneo_ref > 0);
346
347 const float c = (float)eog_centrado;
348 const float arriba_det = FRAC_DETECCION * (float)umbral_arriba_ref;
349 const float arriba_ret = FRAC_RETORNO * (float)umbral_arriba_ref;
350 const float abajo_det = FRAC_DETECCION * (float)umbral_abajo_ref;
351 const float abajo_ret = FRAC_RETORNO * (float)umbral_abajo_ref;
352 const float click_det = FRAC_PESTANEO * (float)umbral_pestaneo_ref;
353
354 // Pestañeo -> click
355 if (pestaneo_ok && estado_actual != ESTADO_BLOQUEO && c > click_det) {
356 *click_flag = 1;
357 EntrarBloqueo(&t_estado, &t_banda);
358 return;
359 }
360
361 switch (estado_actual) {
362 case ESTADO_CENTRO:
363 if (arriba_ok && c >= arriba_det) {
364 estado_actual = ESTADO_ESPERANDO_RETORNO_ARRIBA;
365 t_estado = 0;
366 } else if (abajo_ok && c <= abajo_det) {
367 estado_actual = ESTADO_ESPERANDO_RETORNO_ABAJO;
368 t_estado = 0;
369 }
370 break;
371
372 case ESTADO_ESPERANDO_RETORNO_ARRIBA:
373 t_estado++;
374 if (c < arriba_ret) {
375 *pos = -PASO_MOUSE_PX;
376 EntrarBloqueo(&t_estado, &t_banda);
377 } else if (t_estado > ESPERA_RETORNO_MAX) {
378 EntrarBloqueo(&t_estado, &t_banda);
379 }
380 break;
381
382 case ESTADO_ESPERANDO_RETORNO_ABAJO:
383 t_estado++;
384 if (c > abajo_ret) {
385 *pos = PASO_MOUSE_PX;
386 EntrarBloqueo(&t_estado, &t_banda);
387 } else if (t_estado > ESPERA_RETORNO_MAX) {
388 EntrarBloqueo(&t_estado, &t_banda);
389 }
390 break;
391
392 case ESTADO_BLOQUEO: {
393 const float banda_sup = arriba_ok ? arriba_ret : UMBRAL_DET_MIN_CUENTAS;
394 const float banda_inf = abajo_ok ? abajo_ret : -UMBRAL_DET_MIN_CUENTAS;
395
396 t_estado++;
397 if (c < banda_sup && c > banda_inf) {
398 t_banda++;
399 } else {
400 t_banda = 0;
401 }
402
403 if (t_banda >= REARME_MUESTRAS || t_estado >= BLOQUEO_MAX_MUESTRAS) {
404 estado_actual = ESTADO_CENTRO;
405 }
406 break;
407 }
408 }
409}

430void AdquisicionTask(void *pvParameter) {
431 uint8_t decimation_counter = 0;
432 muestra_eog_t muestra;
433
434 CalibrarEOG();
435 IniciarMuestreo();
436
437 while (true) {
438 EsperarTick();
439
440 uint16_t raw = 0;
441 float filtrada = LeerMuestraFiltrada(&raw);
442 int16_t centrado = (int16_t)(filtrada - (float)offset_v);
443
444 if (++decimation_counter >= DECIMATION_FACTOR) {
445 decimation_counter = 0;
446 muestra.raw = raw;
447 muestra.centrado = centrado;
448 xQueueSend(cola_muestras, &muestra, 0);
449 }
450 }

454void ProcesamientoOutputTask(void *pvParameter) {
455 muestra_eog_t muestra;
456 int8_t mouse_y = 0;
457 uint8_t mouse_click = 0;
458
459 while (true) {
460 if (xQueueReceive(cola_muestras, &muestra, portMAX_DELAY) == pdTRUE) {
461 UpdateMouse(&mouse_y, &mouse_click, muestra.centrado);
462
463 if (mouse_click) {
464 BleHidSendMouse(HID_MOUSE_LEFT, 0, 0);
465 vTaskDelay(pdMS_TO_TICKS(CLICK_PULSO_MS));
466 BleHidSendMouse(0, 0, 0);
467 } else if (mouse_y != 0) {
468 BleHidSendMouse(0, 0, mouse_y);
469 }
470 printf("%d,%d,%u\r\n", muestra.centrado, mouse_y, mouse_click);
471 }
472 }
