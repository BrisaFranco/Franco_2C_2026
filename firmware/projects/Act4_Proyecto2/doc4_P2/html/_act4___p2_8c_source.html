145void app_main(void)
146{
147 /* 1. Inicialización del puerto serie UART */
148 serial_config_t uart_cfg = {
149 .port = UART_PC,
150 .baud_rate = UART_BAUD_RATE,
151 .func_p = NULL,
152 .param_p = NULL
153 };
154 UartInit(&uart_cfg);
155
156 /* 2. Configuración de Entrada Analógica (CH1, lectura puntual) */
157 analog_input_config_t adc_cfg = {
158 .input = CH1,
159 .mode = ADC_SINGLE
160 };
161 AnalogInputInit(&adc_cfg);
162
163 /* 3. Inicialización de Salida Analógica (DAC) */
164 AnalogOutputInit();
165
166 /* 4. Timer A: Muestreo del ADC a 500 Hz (2000 us) */
167 timer_config_t timer_adc = {
168 .timer = TIMER_A,
169 .period = FREC_ADC_US,
170 .func_p = FuncTimerADC,
171 .param_p = NULL
172 };
173 TimerInit(&timer_adc);
174
175 /* 5. Timer B: Generación del DAC a 250 Hz (4000 us) */
176 timer_config_t timer_dac = {
177 .timer = TIMER_B,
178 .period = FREC_DAC_US,
179 .func_p = FuncTimerDAC,
180 .param_p = NULL
181 };
182 TimerInit(&timer_dac);
183
184 /* 6. Creación de Tareas FreeRTOS */
185 xTaskCreate(ADCTask, "ADCTask", 2048, NULL, 5, &main_task_handle);
186 xTaskCreate(DACTask, "DACTask", 2048, NULL, 5, &dac_task_handle);
187
188 /* 7. Inicio de los Timers */
189 TimerStart(timer_adc.timer);
190 TimerStart(timer_dac.timer);
191}
